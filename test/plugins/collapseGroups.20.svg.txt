Collapse a group with a style attribute when none of its declarations
conflict with the child's own styles.

===

<svg xmlns="http://www.w3.org/2000/svg">
    <g style="fill:red">
        <rect width="10" height="10"/>
    </g>
    <g style="fill:red">
        <rect stroke="blue" width="20" height="20"/>
    </g>
</svg>

@@@

<svg xmlns="http://www.w3.org/2000/svg">
    <rect width="10" height="10" style="fill:red"/>
    <rect stroke="blue" width="20" height="20" style="fill:red"/>
</svg>
